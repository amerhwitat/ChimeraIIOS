cmake_minimum_required(VERSION 3.16)
project(ChimeraIIOS VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CHIMERA_ENABLE_EXPERIMENTAL "Build experimental machine/cognitive prototypes" ON)
add_library(chimera_core STATIC src/chimera.c)
target_include_directories(chimera_core PUBLIC include)
target_compile_options(chimera_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(chimera_machine STATIC src/isa/chimera_isa.cpp)
target_include_directories(chimera_machine PUBLIC include)
target_compile_options(chimera_machine PRIVATE -Wall -Wextra -Wpedantic)
add_executable(chimera_server src/server.c)
target_link_libraries(chimera_server PRIVATE chimera_core pthread)
target_compile_options(chimera_server PRIVATE -Wall -Wextra -Wpedantic)
add_executable(chimera_unit tests/unit/test_core.cpp)
target_link_libraries(chimera_unit PRIVATE chimera_machine)
target_include_directories(chimera_unit PRIVATE include)
enable_testing()
add_test(NAME chimera_unit COMMAND chimera_unit)
