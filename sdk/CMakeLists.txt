cmake_minimum_required(VERSION 3.16)
project(ChimeraSDK VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
add_library(chimera_sdk STATIC src/chimera_sdk.cpp)
target_include_directories(chimera_sdk PUBLIC include)
add_executable(chimera-sdk-config tools/chimera-sdk-config.cpp)
target_link_libraries(chimera-sdk-config PRIVATE chimera_sdk)
install(TARGETS chimera_sdk chimera-sdk-config ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY python/chimera_sdk DESTINATION share/chimera/python)
install(DIRECTORY csharp DESTINATION share/chimera/sdk)
install(DIRECTORY objective-c DESTINATION share/chimera/sdk)
install(DIRECTORY java DESTINATION share/chimera/sdk)
install(DIRECTORY manuals DESTINATION share/chimera/sdk)
install(DIRECTORY examples DESTINATION share/chimera/sdk)
install(FILES toolchains/chimera-gcc.cmake toolchains/chimera-clang.cmake DESTINATION share/chimera/toolchains)
