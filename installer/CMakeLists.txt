cmake_minimum_required(VERSION 3.16)
project(ChimeraInstaller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(chimera_installer_core STATIC src/installer_core.cpp)
target_include_directories(chimera_installer_core PUBLIC include)
add_executable(chimera-installer-cli src/installer_cli.cpp)
target_link_libraries(chimera-installer-cli PRIVATE chimera_installer_core)
add_executable(chimera-installer-gui src/installer_gui.cpp)
target_link_libraries(chimera-installer-gui PRIVATE chimera_installer_core)
install(TARGETS chimera-installer-cli chimera-installer-gui RUNTIME DESTINATION bin)
install(DIRECTORY profiles DESTINATION share/chimera/installer)
