cmake_minimum_required(VERSION 3.16)
project(ChimeraFlashTool VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(chimera_flash_tool src/main.cpp)
target_compile_features(chimera_flash_tool PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(chimera_flash_tool PRIVATE /W4 /permissive-)
else()
  target_compile_options(chimera_flash_tool PRIVATE -Wall -Wextra -Wpedantic)
endif()
