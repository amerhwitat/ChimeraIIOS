cmake_minimum_required(VERSION 3.16)
project(ChimeraDatabaseAdapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(chimera_db_backend STATIC src/db_backend.cpp)
target_include_directories(chimera_db_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(chimera_db_backend_test ../tests/unit/test_database_backend.cpp)
target_link_libraries(chimera_db_backend_test PRIVATE chimera_db_backend)
enable_testing()
add_test(NAME chimera_db_backend COMMAND chimera_db_backend_test)
install(TARGETS chimera_db_backend ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
